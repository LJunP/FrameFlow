package com.frameflow.product;

import static org.assertj.core.api.Assertions.assertThat;

import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.frameflow.identity.support.IdentityIntegrationTestBase;
import com.frameflow.product.web.dto.ProductDtos.FindingDto;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardCopyOption;
import java.util.List;
import java.util.Map;
import java.util.UUID;
import java.util.concurrent.TimeUnit;
import org.junit.jupiter.api.Assumptions;
import org.junit.jupiter.api.Test;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.test.context.DynamicPropertyRegistry;
import org.springframework.test.context.DynamicPropertySource;

/** FF-VSL-003 single-candidate vertical slice E2E with the real Python worker. */
class SingleCandidateSliceE2ETest extends IdentityIntegrationTestBase {

    private static final Path WORKER_ROOT = resolveWorkerRoot();
    private static final boolean WORKER_AVAILABLE =
            WORKER_ROOT.resolve(".venv/bin/python").toFile().exists();

    private static Path resolveWorkerRoot() {
        for (String p : new String[]{"../frameflow-ai-worker", "frameflow-ai-worker"}) {
            Path candidate = java.nio.file.Paths.get(p).toAbsolutePath().normalize();
            if (candidate.resolve(".venv/bin/python").toFile().exists()) {
                return candidate;
            }
        }
        return java.nio.file.Paths.get("frameflow-ai-worker").toAbsolutePath();
    }

    private static String workerPython() {
        return WORKER_ROOT.resolve(".venv/bin/python").toString();
    }

    @DynamicPropertySource
    static void workerProperties(DynamicPropertyRegistry registry) {
        if (WORKER_AVAILABLE) {
            registry.add("frameflow.worker.python", SingleCandidateSliceE2ETest::workerPython);
            registry.add("frameflow.worker.cli", () -> "frameflow_ai.cli");
            registry.add("frameflow.storage.root", () -> "data/media-test");
        }
    }

    private static final ObjectMapper MAP = new ObjectMapper();

    private Map<String, Object> body(ResponseEntity<String> resp) {
        return json(resp.getBody());
    }

    private HttpHeaders teamHeader(String token, long teamId) {
        HttpHeaders h = tokenHeaders(token);
        h.set("X-Team-Id", String.valueOf(teamId));
        return h;
    }

    private HttpHeaders keyed(String token, String key) {
        HttpHeaders h = tokenHeaders(token);
        h.set("Idempotency-Key", key);
        return h;
    }

    private String payload() {
        return "{\"schemaVersion\":\"1.0.0\",\"name\":\"q\",\"templateType\":\"ECOMMERCE_SHORT_AD_V1\","
                + "\"inputContract\":{\"durationSeconds\":{\"minimum\":3,\"maximum\":60},\"aspectRatio\":{\"expected\":\"9:16\",\"tolerance\":0.02},\"audioRequired\":true},"
                + "\"automationPolicy\":{\"defaultSemanticAction\":\"REVIEW\",\"semanticAutoRejectEnabled\":false,\"unknownResultAction\":\"REVIEW\"},"
                + "\"rules\":[{\"ruleId\":\"FF-RULE-DURATION\",\"version\":1,\"type\":\"HARD_CONSTRAINT\",\"dimension\":\"technical_quality\","
                + "\"severity\":\"BLOCKER\",\"automationAction\":\"REJECT\",\"evidenceRequirement\":{\"type\":\"METADATA\"}},"
                + "{\"ruleId\":\"FF-RULE-ASPECT\",\"version\":1,\"type\":\"HARD_CONSTRAINT\",\"dimension\":\"technical_quality\","
                + "\"severity\":\"BLOCKER\",\"automationAction\":\"REJECT\",\"evidenceRequirement\":{\"type\":\"METADATA\"}}],"
                + "\"ranking\":{\"dimensions\":{\"prompt_alignment\":1.0},\"unknownDimensionPolicy\":\"REVIEW_REQUIRED\"}}";
    }

    @Test
    void singleCandidateFullSliceWithWorker() throws Exception {
        Assumptions.assumeTrue(WORKER_AVAILABLE, "python worker venv not present; skipping");

        String email = nextEmail();
        register(email, "passw0rd!");
        Map login = login(email, "passw0rd!");
        String token = accessTokenOf(login);
        ResponseEntity<String> teamResp = postJson("/api/v1/teams", Map.of("name", nextName()),
                keyed(token, UUID.randomUUID().toString()));
        assertThat(teamResp.getStatusCode()).isEqualTo(HttpStatus.CREATED);
        ResponseEntity<String> teams = getJson("/api/v1/teams", tokenHeaders(token));
        List<Map<String, Object>> items = jsonList(json(teams.getBody()).get("items"));
        long teamId = ((Number) items.get(0).get("teamId")).longValue();
        HttpHeaders h = teamHeader(token, teamId);

        ResponseEntity<String> prj = postJson("/api/v1/projects",
                Map.of("name", "slice-e2e", "description", "worker vertical slice"), h);
        long projectId = ((Number) body(prj).get("id")).longValue();

        ResponseEntity<String> prof = postJson("/api/v1/projects/" + projectId + "/quality-profiles",
                Map.of("name", "q", "templateType", "ECOMMERCE_SHORT_AD_V1"), h);
        long profileId = ((Number) body(prof).get("id")).longValue();
        ResponseEntity<String> ver = postJson("/api/v1/projects/" + projectId + "/quality-profiles/" + profileId + "/versions",
                Map.of("payload", payload()), h);
        long versionId = ((Number) body(ver).get("id")).longValue();
        ResponseEntity<String> pub = postJson("/api/v1/quality-profile-versions/" + versionId + "/publish", Map.of(), tokenHeaders(token));
        assertThat(body(pub).get("status")).isEqualTo("PUBLISHED");

        ResponseEntity<String> batch = postJson("/api/v1/projects/" + projectId + "/batches",
                Map.of("name", "b", "profileVersionId", versionId), h);
        long batchId = ((Number) body(batch).get("id")).longValue();

        ResponseEntity<String> cands = postJson("/api/v1/batches/" + batchId + "/candidates",
                Map.of("keys", List.of("slice-1"), "mediaTypes", List.of("video/mp4")), h);
        List<Long> ids = MAP.readValue(cands.getBody(), new TypeReference<List<Long>>() {
        });
        long candidateId = ids.get(0);

        ResponseEntity<String> up = postJson("/api/v1/candidates/" + candidateId + "/upload-session", Map.of(), h);
        long sessionId = ((Number) body(up).get("sessionId")).longValue();
        ResponseEntity<String> complete = postJson("/api/v1/candidates/" + candidateId + "/upload-complete",
                Map.of("sessionId", sessionId, "sizeBytes", 100000L, "contentDigest", "abc"), h);
        assertThat(body(complete).get("status")).isEqualTo("STORED");

        Path fixture = Path.of("experiments/fixtures/generated/normal_vertical.mp4");
        Assumptions.assumeTrue(fixture.toFile().exists(), "fixture missing; skipping");
        Path stored = Path.of("data/media-test/teams/" + teamId + "/batches/" + batchId + "/candidates/"
                + candidateId + "/v1/source.mp4");
        Files.createDirectories(stored.getParent());
        Files.copy(fixture, stored, StandardCopyOption.REPLACE_EXISTING);

        ResponseEntity<String> start = postJson("/api/v1/batches/" + batchId + "/analysis-runs",
                Map.of("candidateId", candidateId), h);
        assertThat(start.getStatusCode()).isEqualTo(HttpStatus.ACCEPTED);
        long runId = ((Number) body(start).get("runId")).longValue();
        String runCommand = String.valueOf(body(start).get("commandId"));

        Path profileTmp = Files.createTempFile("ff-profile-", ".json");
        Files.writeString(profileTmp, payload());
        ProcessBuilder pb = new ProcessBuilder(workerPython(), "-m", "frameflow_ai.cli", "analyze",
                stored.toString(), profileTmp.toString());
        pb.redirectErrorStream(true);
        Process p = pb.start();
        String out = new String(p.getInputStream().readAllBytes(), java.nio.charset.StandardCharsets.UTF_8);
        assertThat(p.waitFor(180, TimeUnit.SECONDS)).isTrue();
        assertThat(p.exitValue()).isZero();

        JsonNode doc = MAP.readTree(out);
        String status = doc.path("status").asText();
        String decisionJson = MAP.writeValueAsString(doc.path("decision"));
        List<Map<String, Object>> rawFindings = MAP.readValue(doc.path("findings").toString(),
                new TypeReference<List<Map<String, Object>>>() {
                });
        List<FindingDto> dtos = rawFindings.stream().map(m -> new FindingDto(
                (String) m.get("ruleId"), null, (String) m.get("detectorId"),
                (String) m.get("detectorVersion"), (String) m.get("dimension"), null,
                (String) m.get("verdict"), (String) m.get("severity"),
                m.get("confidence") == null ? null : ((Number) m.get("confidence")).doubleValue(),
                (String) m.get("automationAction"),
                m.get("startMs") == null ? null : ((Number) m.get("startMs")).intValue(),
                m.get("endMs") == null ? null : ((Number) m.get("endMs")).intValue(),
                (String) m.get("summary"), null, null)).toList();

        ResponseEntity<String> ingest = postJson("/api/v1/internal/analysis-runs/" + runId + "/results",
                Map.of("commandId", runCommand, "status", status, "decision", decisionJson,
                        "qualityVector", "{}", "findings", dtos), h);
        assertThat(ingest.getStatusCode()).isEqualTo(HttpStatus.NO_CONTENT);

        ResponseEntity<String> runResp = getJson("/api/v1/analysis-runs/" + runId, tokenHeaders(token));
        assertThat(body(runResp).get("status")).isEqualTo("COMPLETED");
        ResponseEntity<String> findingsResp = getJson("/api/v1/candidates/" + candidateId + "/findings",
                tokenHeaders(token));
        assertThat(findingsResp.getStatusCode()).isEqualTo(HttpStatus.OK);
        List<Map<String, Object>> saved = MAP.readValue(findingsResp.getBody(),
                new TypeReference<List<Map<String, Object>>>() {
                });
        assertThat(saved).isNotEmpty();
        assertThat(saved.stream().anyMatch(m -> "FF-RULE-DURATION".equals(m.get("ruleId")))).isTrue();
    }
}
